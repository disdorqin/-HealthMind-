#### 前端层面

import pandas as pd
import numpy as np

def create_time_features(df, date_col='date'):
    """创建时间特征"""
    df[date_col] = pd.to_datetime(df[date_col])
    
    # 基本时间特征
    df['year'] = df[date_col].dt.year
    df['month'] = df[date_col].dt.month
    df['day'] = df[date_col].dt.day
    df['dayofweek'] = df[date_col].dt.dayofweek
    df['quarter'] = df[date_col].dt.quarter
    df['weekofyear'] = df[date_col].dt.isocalendar().week
    
    # 周期性特征（正弦余弦编码）
    df['month_sin'] = np.sin(2 * np.pi * df['month'] / 12)
    df['month_cos'] = np.cos(2 * np.pi * df['month'] / 12)
    df['dayofweek_sin'] = np.sin(2 * np.pi * df['dayofweek'] / 7)
    df['dayofweek_cos'] = np.cos(2 * np.pi * df['dayofweek'] / 7)
    
    return df

def create_lag_features(df, target_col, lags=[1, 7, 30]):
    """创建滞后特征"""
    for lag in lags:
        df[f'{target_col}_lag_{lag}'] = df[target_col].shift(lag)
    return df

def create_rolling_features(df, target_col, windows=[7, 30]):
    """创建滚动统计特征"""
    for window in windows:
        df[f'{target_col}_rolling_mean_{window}'] = df[target_col].rolling(window).mean()
        df[f'{target_col}_rolling_std_{window}'] = df[target_col].rolling(window).std()
    return df
```

# 数据质量检查脚本
import pandas as pd
import numpy as np

def data_quality_report(df):
    """生成数据质量报告"""
    report = {
        '总样本数': len(df),
        '特征数量': len(df.columns),
        '缺失值统计': df.isnull().sum().to_dict(),
        '重复行数量': df.duplicated().sum(),
        '数值特征统计': df.describe().to_dict(),
    }
    
    # 异常值检测（3σ原则）
    for col in df.select_dtypes(include=[np.number]).columns:
        mean, std = df[col].mean(), df[col].std()
        outliers = ((df[col] - mean) > 3 * std).sum()
        report[f'{col}_异常值数量'] = outliers
    
    return report

# 使用示例
df = pd.read_csv("data/personal_carbon_footprint_behavior.csv")
report = data_quality_report(df)
print(report)

# 新建 src/models/attention_lstm.py
import torch
import torch.nn as nn

class Attention(nn.Module):
    """注意力机制"""
    def __init__(self, hidden_dim):
        super().__init__()
        self.attention = nn.Linear(hidden_dim, 1)
    
    def forward(self, lstm_output):
        # lstm_output: (batch, seq_len, hidden_dim)
        attention_scores = self.attention(lstm_output)  # (batch, seq_len, 1)
        attention_weights = torch.softmax(attention_scores, dim=1)
        context = torch.sum(attention_weights * lstm_output, dim=1)
        return context

class AttentionLSTM(nn.Module):
    """带注意力的 LSTM"""
    def __init__(self, input_dim, hidden_dim, num_layers):
        super().__init__()
        self.lstm = nn.LSTM(
            input_size=input_dim,
            hidden_size=hidden_dim,
            num_layers=num_layers,
            batch_first=True,
            bidirectional=True
        )
        self.attention = Attention(hidden_dim * 2)
        self.head = nn.Linear(hidden_dim * 2, 1)
    
    def forward(self, x):
        lstm_output, _ = self.lstm(x)
        context = self.attention(lstm_output)
        return self.head(context).squeeze(-1)
```

### 4.2 模型创新

#### 方案 A：混合模型（推荐 ⭐⭐⭐⭐）

**时序 + 特征双通道模型**：
```
原始特征 ──→ XGBoost 分支 ──┐
                          ├──→ 融合层 ──→ 预测结果
时序数据 ──→ LSTM 分支 ────┘
```

**实施代码**（新建 `src/models/hybrid_model.py`）：
```python
import torch
import torch.nn as nn
import numpy as np
from xgboost import XGBRegressor

class HybridModel(nn.Module):
    """混合模型：LSTM + XGBoost"""
    
    def __init__(self, input_dim, hidden_dim=128, num_layers=2):
        super().__init__()
        
        # LSTM 分支
        self.lstm = nn.LSTM(
            input_size=input_dim,
            hidden_size=hidden_dim,
            num_layers=num_layers,
            batch_first=True,
            bidirectional=True
        )
        
        # 融合层
        self.fusion = nn.Linear(hidden_dim * 2 + 1, 1)
        
        # XGBoost（用于特征分支）
        self.xgb = XGBRegressor(n_estimators=100, max_depth=6)
    
    def forward(self, x_seq, x_flat):
        """
        x_seq: 时序数据 (batch, seq_len, features)
        x_flat: 扁平特征 (batch, features)
        """
        # LSTM 分支
        lstm_out, _ = self.lstm(x_seq)
        lstm_feat = lstm_out[:, -1, :]  # 取最后一步
        
        # XGBoost 分支
        xgb_pred = self.xgb.predict(x_flat.cpu().numpy())
        xgb_feat = torch.tensor(xgb_pred).float().unsqueeze(-1)
        
        # 融合
        combined = torch.cat([lstm_feat, xgb_feat], dim=1)
        output = self.fusion(combined)
        
        return output.squeeze(-1)
```

#### 方案 B：集成学习改进（推荐 ⭐⭐⭐⭐⭐）

**当前问题**：简单平均未考虑模型性能差异

**改进方案**：性能加权集成
```python
# 修改 src/models/stacking_manager.py
class ImprovedStackingManager:
    """改进的 Stacking 管理器"""
    
    def __init__(self):
        self.model_weights = {}
        self.model_history = {}
    
    def calculate_weights(self, model_predictions, ground_truth):
        """基于历史性能计算权重"""
        weights = {}
        
        for model_name, preds in model_predictions.items():
            # 计算 R² 分数
            ss_res = np.sum((ground_truth - preds) ** 2)
            ss_tot = np.sum((ground_truth - np.mean(ground_truth)) ** 2)
            r2 = 1 - (ss_res / ss_tot)
            
            # R² 转换为权重（只考虑正 R²）
            if r2 > 0:
                weights[model_name] = r2
            else:
                weights[model_name] = 0.01  # 负 R² 给很小权重
        
        # 归一化
        total = sum(weights.values())
        weights = {k: v / total for k, v in weights.items()}
        
        return weights
    
    def predict(self, model_predictions):
        """性能加权预测"""
        weights = self.calculate_weights(
            model_predictions, 
            self.last_ground_truth
        )
        
        ensemble = np.zeros(len(list(model_predictions.values())[0]))
        for model_name, preds in model_predictions.items():
            ensemble += preds * weights.get(model_name, 0)
        
        return ensemble, weights
```

from sklearn.model_selection import cross_val_score

def optimize_lstm(X_train, y_train, X_val, y_val, n_trials=50):
    """LSTM 超参数优化"""
    
    def objective(trial):
        # 定义搜索空间
        hidden_dim = trial.suggest_categorical('hidden_dim', [64, 128, 256, 512])
        num_layers = trial.suggest_int('num_layers', 1, 4)
        learning_rate = trial.suggest_float('learning_rate', 1e-4, 1e-2, log=True)
        dropout = trial.suggest_float('dropout', 0.1, 0.5)
        batch_size = trial.suggest_categorical('batch_size', [16, 32, 64])
        
        # 创建模型
        model = LSTMForecastModel(
            input_dim=X_train.shape[2],
            hidden_dim=hidden_dim,
            num_layers=num_layers,
            learning_rate=learning_rate
        )
        
        # 训练
        history = model.train(
            X_train, y_train, X_val, y_val,
            epochs=50, batch_size=batch_size
        )
        
        # 返回验证集 loss
        return min(history['val_loss'])

    study = optuna.create_study(direction='minimize')
    study.optimize(objective, n_trials=n_trials)
    
    print("最佳参数：", study.best_params)

```python
import sqlite3
from datetime import datetime

class UserDatabase:
    """用户数据库"""
    
    def __init__(self, db_path="database/ecolife.db"):
        self.db_path = db_path
        self._init_tables()
    
    def _init_tables(self):
        """初始化表结构"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        # 用户表
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS users (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                username TEXT UNIQUE,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        ''')
        
        # 碳足迹记录表
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS carbon_records (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_id INTEGER,
                date DATE,
                carbon_value REAL,
                predicted_value REAL,
                FOREIGN KEY (user_id) REFERENCES users (id)
            )
        ''')
        
        # 积分表
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS credits (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_id INTEGER,
                credits INTEGER,
                reason TEXT,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (user_id) REFERENCES users (id)
            )
        ''')
        
        conn.commit()
        conn.close()
    
    def add_user(self, username):
        """添加用户"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        try:
            cursor.execute("INSERT INTO users (username) VALUES (?)", (username,))
            conn.commit()
            return cursor.lastrowid
        except sqlite3.IntegrityError:
            return None
        finally:
            conn.close()
    
    def add_carbon_record(self, user_id, date, carbon_value, predicted_value=None):
        """添加碳足迹记录"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute(
            "INSERT INTO carbon_records (user_id, date, carbon_value, predicted_value) VALUES (?, ?, ?, ?)",
            (user_id, date, carbon_value, predicted_value)
        )
        conn.commit()
        conn.close()
    
    def get_user_history(self, user_id, days=30):
        """获取用户历史记录"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute(
            "SELECT date, carbon_value, predicted_value FROM carbon_records WHERE user_id = ? ORDER BY date DESC LIMIT ?",
            (user_id, days)
        )
        records = cursor.fetchall()
        conn.close()
        return records
```

```python
from pyecharts.charts import Line
from pyecharts import options as opts

def build_forecast_option(x_axis, curves, selected, title):
    line = Line()
    line.add_xaxis(x_axis)
    
    for name, values in curves.items():
        line.add_yaxis(
            series_name=name,
            y_axis=[float(v) for v in values],
            is_smooth=True,  # 平滑曲线
            is_symbol_show=False,  # 不显示数据点
            is_selected=name in selected,  # 默认显示/隐藏
            linestyle_opts=opts.LineStyleOpts(width=2.6 if name == "ensemble" else 1.8),
            itemstyle_opts=opts.ItemStyleOpts(color=palette.get(name, "#6b7280")),
        )
    
    line.set_global_opts(
        title_opts=opts.TitleOpts(title=title),
        legend_opts=opts.LegendOpts(type_="scroll"),  # 可滚动图例
        tooltip_opts=opts.TooltipOpts(trigger="axis"),
        datazoom_opts=[opts.DataZoomOpts(type_="inside"), opts.DataZoomOpts(type_="slider")],
    )
    
    return json.loads(line.dump_options())
```

**图表配置说明：**

| 配置项 | 值 | 说明 |
|--------|-----|------|
| `is_smooth` | `True` | 平滑曲线，提升视觉体验 |
| `is_symbol_show` | `False` | 隐藏数据点标记，避免密集 |
| `type_="scroll"` | - | 可滚动图例，支持多模型 |
| `datazoom_opts` | `inside+slider` | 内部缩放 + 底部滑块 |

```python
from sklearn.linear_model import Ridge

class StackingManager:
    def __init__(self, alpha=1.0):
        self.meta_learner = Ridge(alpha=alpha, random_state=42)
    
    def train(self, base_predictions, y_true):
        # 构造元特征矩阵
        X_meta = np.column_stack([base_predictions[k].reshape(-1) 
                                   for k in sorted(base_predictions.keys())])
        self.meta_learner.fit(X_meta, y_true)
    
    def predict(self, base_predictions):
        X_meta = np.column_stack([base_predictions[k].reshape(-1) 
                                   for k in sorted(base_predictions.keys())])
        return self.meta_learner.predict(X_meta)
```

```python
import pandas as pd
import numpy as np

def load_and_engineer_features(csv_path, target_col="YD15"):
    df = pd.read_csv(csv_path)
    
    # 时间特征工程
    if "DATATIME" in df.columns:
        dt = pd.to_datetime(df["DATATIME"], errors="coerce")
        df["hour"] = dt.dt.hour.fillna(0).astype(int)
        df["dayofweek"] = dt.dt.dayofweek.fillna(0).astype(int)
        df["month"] = dt.dt.month.fillna(1).astype(int)
    
    # 缺失值处理
    numeric_df = df.select_dtypes(include=[np.number]).copy()
    numeric_df = numeric_df.replace([np.inf, -np.inf], np.nan).ffill().bfill().fillna(0.0)
    
    # 特征矩阵
    X = numeric_df.drop(columns=[target_col]).to_numpy(dtype=np.float32)
    y = numeric_df[target_col].to_numpy(dtype=np.float32)
    
    return X, y, df
```

```python
import logging

def setup_logger(name="fengmang"):
    logger = logging.getLogger(name)
    logger.setLevel(logging.INFO)
    
    # 控制台处理器
    console_handler = logging.StreamHandler()
    console_handler.setLevel(logging.INFO)
    
    # 文件处理器
    file_handler = logging.FileHandler("logs/app.log")
    file_handler.setLevel(logging.DEBUG)
    
    # 格式化器
    formatter = logging.Formatter(
        "%(asctime)s - %(name)s - %(levelname)s - %(message)s"
    )
    console_handler.setFormatter(formatter)
    file_handler.setFormatter(formatter)
    
    logger.addHandler(console_handler)
    logger.addHandler(file_handler)
    
    return logger

logger = setup_logger()
```

| 云端部署 | 原生支持 | 需配置 | Streamlit Cloud 一键部署 |

| 维度 | MySQL | PostgreSQL | SQLite | 选择理由 |
|------|-------|------------|--------|----------|
| 性能 | 高 | 高 | 中 | 满足并发需求 |
| 学习成本 | 低 | 中 | 低 | 团队熟悉度高 |
| 部署便捷 | 高 | 中 | 高 | Docker 镜像丰富 |
| 社区支持 | 强 | 强 | 强 | 问题易解决 |

### 5.9.3 为何选择 PyTorch 而非 TensorFlow？

| 维度 | PyTorch | TensorFlow | 选择理由 |
|------|---------|------------|----------|
| 易用性 | 高 | 中 | Pythonic API |
| 调试体验 | 好 | 一般 | 动态图易调试 |
| 学术社区 | 强 | 中 | 论文代码多用 |
| 部署能力 | 中 | 高 | 竞赛场景够用 |
